Topics:
  ros2 topic echo /battery/state        # Battery state
  ros2 topic echo /battery/low          # Low battery flag
  ros2 topic echo /mission/command      # Mission commands
  ros2 topic echo /zones --once         # All zones
  ros2 topic echo /zone/current --once  # Active zone

Services:
  ros2 service call /zone/list rosmower_msgs/srv/ListZones
  ros2 service call /zone/load rosmower_msgs/srv/LoadZone "{zone_id: 'front_yard'}"
  ros2 service call /zone/delete rosmower_msgs/srv/DeleteZone "{zone_id: 'test'}"

Test Battery:
  ros2 topic pub /percent std_msgs/msg/Float32 "data: 100.0"  # Normal
  ros2 topic pub /percent std_msgs/msg/Float32 "data: 20.0"   # Low
  ros2 topic pub /percent std_msgs/msg/Float32 "data: 10.0"   # Critical
